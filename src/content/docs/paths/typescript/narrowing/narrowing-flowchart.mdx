## Tools at your disposal

```ts
function f(x: string | number | Date | null) {
  if (x === null) return;                // equality
  if (typeof x === 'number') /* number */;
  if (typeof x === 'string') /* string */;
  if (x instanceof Date) /* Date */;
}

function g(x: { kind: 'a'; a: string } | { kind: 'b'; b: number }) {
  if (x.kind === 'a') /* x is the 'a' variant */;
  // discriminated union
}

function h(x: object) {
  if ('foo' in x) /* x has 'foo' */;
}

function i(x: unknown) {
  if (Array.isArray(x)) /* x is unknown[] */;
}
```

## Truthiness

```ts
function j(x: string | undefined) {
  if (x) /* x is string (and not '') */;
  if (x !== undefined) /* x is string (including '') */;
}
```

`if (x)` excludes empty string AND undefined. Sometimes you want only undefined excluded.

## After narrowing

Once narrowed, the type sticks until the next branch:

```ts
function k(x: string | null) {
  if (x === null) return;
  // here, x is string
  x.toUpperCase(); // ✓
}
```
